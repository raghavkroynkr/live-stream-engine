name: Universal Video Downloader on: workflow_dispatch: inputs: video_url: required: true email: required: true format: default: 'best' jobs: download: runs-on: ubuntu-latest steps: - name: Install Latest yt-dlp run: | python3 -m pip install --upgrade yt-dlp - name: Download Video (Bypass Mode) run: | # Ismein hum user-agent aur sleep time use kar rahe hain taaki bot na pakde if [ "${{ github.event.inputs.format }}" = "mp3" ]; then yt-dlp --user-agent "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/114.0.0.0 Safari/537.36" \ --no-check-certificates --extract-audio --audio-format mp3 -o "downloaded_file.%(ext)s" "${{ github.event.inputs.video_url }}" else yt-dlp --user-agent "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/114.0.0.0 Safari/537.36" \ --no-check-certificates -f "best" -o "downloaded_file.%(ext)s" "${{ github.event.inputs.video_url }}" fi - name: Send via Email shell: python env: MAIL_USER: ${{ secrets.EMAIL_USER }} MAIL_PASS: ${{ secrets.EMAIL_PASS }} TARGET_EMAIL: ${{ github.event.inputs.email }} run: | import smtplib, os from email.message import EmailMessage files = [f for f in os.listdir('.') if f.startswith('downloaded_file')] if not files: print("Download failed, file not found") exit(1) msg = EmailMessage() msg["Subject"] = "Video Download Result" msg["From"] = os.environ["MAIL_USER"] msg["To"] = os.environ["TARGET_EMAIL"] msg.set_content("Aapka video niche attachment mein hai.") with open(files[0], 'rb') as f: msg.add_attachment(f.read(), maintype='application', subtype='octet-stream', filename=files[0]) with smtplib.SMTP_SSL("smtp.gmail.com", 465) as smtp: smtp.login(os.environ["MAIL_USER"], os.environ["MAIL_PASS"]) smtp.send_message(msg)
